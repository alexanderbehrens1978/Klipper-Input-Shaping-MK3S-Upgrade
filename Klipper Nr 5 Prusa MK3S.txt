[gcode_macro _TOOLHEAD_PARK_PAUSE_CANCEL]
description: Helper: park toolhead used in PAUSE and CANCEL_PRINT
variable_extrude: 1.0
gcode:
  ##### set park positon for x and y #####
  # default is your max posion from your printer.cfg
  {% set x_park = printer.toolhead.axis_maximum.x|float - 5.0 %}
  {% set y_park = printer.toolhead.axis_maximum.y|float - 5.0 %}
  {% set z_park_delta = 2.0 %}
  ##### calculate save lift position #####
  {% set max_z = printer.toolhead.axis_maximum.z|float %}
  {% set act_z = printer.toolhead.position.z|float %}
  {% if act_z < (max_z - z_park_delta) %}
    {% set z_safe = z_park_delta %}
  {% else %}
    {% set z_safe = max_z - act_z %}
  {% endif %}
  ##### end of definitions #####
  {% if printer.extruder.can_extrude|lower == 'true' %}
    M83
    G1 E-{extrude} F2100
    {% if printer.gcode_move.absolute_extrude |lower == 'true' %} M82 {% endif %}
  {% else %}
    {action_respond_info("Extruder not hot enough")}
  {% endif %}
  {% if "xyz" in printer.toolhead.homed_axes %}
    G91
    G1 Z{z_safe} F900
    G90
    G1 X{x_park} Y{y_park} F6000
    {% if printer.gcode_move.absolute_coordinates|lower == 'false' %} G91 {% endif %}
  {% else %}
    {action_respond_info("Printer not homed")}
  {% endif %}

[gcode_macro PRINT_START]
gcode:
    # Parameter aus dem Slicer laden (Standardwerte falls nicht übergeben)
    {% set BED_TEMP = params.BED_TEMP|default(60)|float %}
    {% set EXTRUDER_TEMP = params.EXTRUDER_TEMP|default(200)|float %}

    # Homing und Heizen
    G90 ; Absolute Positionierung
    M104 S{EXTRUDER_TEMP} ; Extruder vorheizen
    M190 S{BED_TEMP} ; Bett auf Zieltemperatur bringen und warten
    G28 ; Alle Achsen referenzieren (Home)

    # Bett-Mesh (optional)
    BED_MESH_CALIBRATE

    M109 S{EXTRUDER_TEMP} ; Warten bis Extruder Zieltemperatur erreicht hat

    # -------------------------------------------------------------------
    # PURGE LINE / PUPSLINIE START
    # -------------------------------------------------------------------
    G92 E0 ; Extruder-Position auf 0 setzen

    # Anfahrt: Düse bleibt sicher auf Z=1.5mm während der Fahrt zum Startpunkt
    G1 Z1.5 F3000 ; Düse auf 1.5 mm Sicherheitsabstand anheben
    G1 X0 Y0 F3000 ; Auf Z=1.5 mm schwebend an den vorderen linken Bettrand fahren

    # Erst am Startpunkt der Purge Line auf Druckhöhe absenken
    G1 Z0.2 F720 ; Düse auf 0.2 mm Höhe für die Purge Line absenken

    # Erste Bahn von links nach rechts (X0 -> X60) mit leicht erhöhter Extrusion
    G1 X60 E9.0 F1000 ; Erste Linie ziehen

    # Zweite Bahn von X60 nach X100 mit regulärer Extrusion
    G1 X100 E12.5 F1000 ; Zweite Linie ziehen

    G92 E0 ; Extruder-Position zurücksetzen
    G1 Z2.0 F3000 ; Düse nach der Linie wieder anheben
    # -------------------------------------------------------------------
    # PURGE LINE / PUPSLINIE ENDE
    # -------------------------------------------------------------------
